## Quick Start
```ts
import { StealthSDK } from 'stealthsdk'

const stealth = new StealthSDK({ gameId: 'ares1', token: 'STEALTH' })

await stealth.init({ gameId: 'ares1', token: 'STEALTH' }) // framework AI-games
// economy centralized with STEALTH token
await stealth.createEconomy({ gameId: 'ares1', token: 'STEALTH', supply: 1000000 })
```
